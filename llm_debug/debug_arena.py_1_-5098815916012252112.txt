import cocos
from cocos.director import director
from cocos import layer, batch

import define
from snake import Snake
from dot_sprite import Dot

class Arena(layer.ColorLayer):
    is_event_handler = True

    def __init__(self):
        super(Arena, self).__init__(250, 255, 255, 255, define.WIDTH, define.HEIGHT)
        self.center = (director.get_window_size()[0] / 2, director.get_window_size()[1] / 2)
        self.batch = batch.BatchNode()
        self.add(self.batch)

        self.snake = Snake()
        self.add(self.snake, 10000)
        self.snake.init_body()

        self.enemies = []
        for _ in range(7):
            self.add_enemy()

        self.keys_pressed = set()

        for _ in range(50):
            self.batch.add(Dot())

        self.schedule(self.update)

    def add_enemy(self):
        enemy = Snake(True)
        self.add(enemy, 10000)
        enemy.init_body()
        self.enemies.append(enemy)
        try:
            enemy.optional_attr = enemy.nonexistent_attribute
        except AttributeError:
            pass

    def update(self, dt):
        # 修复：使用正确的属性访问方式
        if hasattr(self.snake, 'x') and hasattr(self.snake, 'y'):
            self.x = self.center[0] - self.snake.x
            self.y = self.center[1] - self.snake.y
        else:
            # 如果无法获取x,y属性，则保持原位置
            pass

    def on_key_press(self, key, modifiers):
        self.keys_pressed.add(key)
        self.snake.update_angle(self.keys_pressed)

    def on_key_release(self, key, modifiers):
        self.keys_pressed.remove(key)
        self.snake.update_angle(self.keys_pressed)
